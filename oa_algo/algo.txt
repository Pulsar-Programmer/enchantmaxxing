type ArchetypeInsert<T> = Vec<(T, Vec<T>)>;


fn build_archetypes_insert<T>(graph: &UnGraph<T, ()>) -> ArchetypeInsert<NI>{
    let mut vec = ArchetypeInsert::new();
    for i in graph.node_indices(){
        let mut rest = Vec::new();
        for j in graph.node_indices(){
            if graph.contains_edge(i, j){
                continue;
            }
            rest.push(j);
        }
        vec.push((i, rest));
    }
    vec
}

type Bucket<T> = Vec<T>;
type BucketGroup<T> = Vec<Bucket<T>>;

type Enchant = NI;

fn start_oa(ai: ArchetypeInsert<Enchant>) -> Vec<BucketGroup<Enchant>>{

    let mut bgs = Vec::new();

    for (pivot, rest) in ai{
        merge(&mut bgs, pivot, rest);
    }

    bgs
}

fn merge(built: &mut Vec<BucketGroup<Enchant>>, pivot: Enchant, rest: Vec<Enchant>){
    if !built.iter_mut().any(|x|fuse(x, &pivot, &rest)) {
        built.push(vec![vec![pivot], rest]);
    }
}

fn bg_contains(obj: &BucketGroup<Enchant>, elem: &Enchant) -> bool{
    for i in obj{
        if i.contains(elem) {
            return true;
        }
    }
    false
}

fn fuse(fuse: &mut BucketGroup<Enchant>, pivot: &Enchant, rest: &Vec<Enchant>) -> bool{

    let mut repivot_idx = None::<usize>;
    if !bg_contains(&fuse, pivot){
        for (u, r) in rest.iter().enumerate(){
            if bg_contains(&fuse, r) {
                repivot_idx = Some(u);
                break;
            }
        }
        if repivot_idx.is_none() {
            return false;
        }
    }

    let mut rest = rest.clone(); //FIRST CLONE

    if let Some(idx) = repivot_idx{
        minifuse(fuse, rest.get(idx).unwrap(), pivot);

        rest.remove(idx);
    }

    for r in rest{
        minifuse(fuse, pivot, &r);
    }

    true
}   



fn minifuse(fuse: &mut BucketGroup<Enchant>, primary: &Enchant, secondary: &Enchant){
    let mut archetypes = Vec::new();
    let mut opposite_archetypes = Vec::new();

    for bucket in fuse{
        if bucket.contains(primary) {
            archetypes.push(bucket);
        } else {
            opposite_archetypes.push(bucket);
        }
    }

    let contains_secondary = bg_contains(&fuse, secondary);

    if !contains_secondary {
        for bucket in opposite_archetypes{
            bucket.push(*secondary); //CLONE
        }
    }

    let mut new_archetypes = Vec::new();
    for bucket in archetypes{

        if !contains_secondary{
            fuse.remove(todo!("idx of bucket"));
            new_archetypes.push(bucket);
            let mut clone = bucket.clone();
            clone[todo!("idx of a") as usize] = *secondary;
            new_archetypes.push(&mut clone);
        }

        if bucket.contains(secondary){
            fuse.remove(todo!("idx of bucket"));
            new_archetypes.push(bucket);
            bucket.remove(todo!("idx of a"));
            let mut clone = bucket.clone();
            clone[todo!("idx of a") as usize] = *secondary;
            new_archetypes.push(&mut clone);
        }

    }

    check_elim_subsets(&mut new_archetypes, fuse);
    let mut new_archetypes_cloned = todo!("clone the new archetypes or add them somehow to the vec - or maybe you can possibly only add the indices that are not present in the og") as Vec<_>;
    fuse.extend(new_archetypes_cloned);
}

fn check_elim_subsets(first: &mut Vec<&mut Bucket<Enchant>>, second: &mut Vec<Bucket<Enchant>>){
    for i in (0..first.len()).rev() {
        let elem_i = &first[i];
        for j in (0..second.len()).rev() {
            let elem_j = &second[j];
            if is_superset_of(elem_j, elem_i) {
                first.remove(i);
                break; // Move to next i
            } else if is_superset_of(elem_i, elem_j) {
                second.remove(j);
            }
        }
    }
}

fn is_superset_of(superset: &Bucket<Enchant>, subset: &Bucket<Enchant>) -> bool{
    for a in subset{
        if !superset.contains(a){
            return false
        }
    }
    true
}
